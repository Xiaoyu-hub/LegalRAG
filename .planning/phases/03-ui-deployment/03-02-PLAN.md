### Step A: Add `generate_stream` method to Generator

In `backend/generation/generator.py`, add a streaming method:

```python
def generate_stream(self, system_prompt: str, user_prompt: str):
    """Stream tokens from Ollama. Yields text tokens as they arrive."""
    models_to_try = [self.model, self.fallback]
    for model in models_to_try:
        try:
            response = chat(
                model=model,
                messages=[
                    {"role": "system", "content": system_prompt},
                    {"role": "user", "content": user_prompt},
                ],
                options={"temperature": 0.1, "num_ctx": 4096, "num_predict": 1024},
                stream=True,
            )
            self._last_model_used = model
            for part in response:
                content = part.get("message", {}).get("content", "")
                if content:
                    yield content
            return
        except ResponseError as e:
            if e.status_code == 404:
                if model == self.fallback:
                    raise RuntimeError(f"Neither {self.model} nor {self.fallback} available")
                continue
            raise
        except ConnectionError:
            raise ConnectionError("Cannot connect to Ollama (ollama serve)")
```

### Step B: Add `answer_stream` method to GenerationPipeline

In `backend/generation/pipeline.py`, add a streaming answer method:

```python
def answer_stream(self, query: str):
    """Stream answer tokens, then yield source data."""
    chunks = []
    if self.retriever is not None and self.metadata_store is not None:
        results = self.retriever.retrieve(query, top_k=5)
        chunk_ids = [r[0] for r in results] if isinstance(results[0], tuple) else results
        chunks = self.metadata_store.get_chunks(chunk_ids)
    self.last_chunks = chunks

    system_prompt, user_prompt = self.prompt_builder.build(query, chunks)
    for token in self.generator.generate_stream(system_prompt, user_prompt):
        yield token
```

The `last_chunks` attribute will be read by the API to return chunk data.

@router.post("/")
async def chat(req: ChatRequest, background_tasks: BackgroundTasks, request: Request):
    from fastapi.responses import StreamingResponse
    from backend.retrieval.hybrid_search import HybridSearch
    from backend.generation.pipeline import GenerationPipeline
    
    store = request.app.state.metadata_store
    
    # Build GenerationPipeline with app state singletons
    retriever = HybridSearch()
    pipeline = GenerationPipeline(retriever=retriever, metadata_store=store)
    
    async def stream_answer():
        # Stream answer tokens
        full_text = ""
        for token in pipeline.answer_stream(req.question):
            full_text += token
            yield token
        
        # Store sources for retrieval by Streamlit
        request.app.state.last_chat_sources = [
            {
                "article": c.article,
                "section": c.section,
                "text": c.text[:500],
                "chunk_id": c.id,
            }
            for c in pipeline.last_chunks
        ]
    
    return StreamingResponse(stream_answer(), media_type="text/plain")
```

import streamlit as st
import requests
import json

import os
API_BASE = os.environ.get("API_BASE") or st.secrets.get("api_base", "http://localhost:8000")

st.title("Chat")

# Get selected document from sidebar (set in main.py)
doc_id = st.session_state.get("selected_doc_id")
doc_name = st.session_state.get("selected_doc_name", "")

if not doc_id:
    st.info("Select a document from the sidebar to begin asking questions")
    st.stop()

# Initialize chat history
if "messages" not in st.session_state:
    st.session_state.messages = []

# Display chat history
for msg in st.session_state.messages:
    with st.chat_message(msg["role"]):
        st.markdown(msg["content"])
        # If assistant message, show sources expander
        if msg["role"] == "assistant" and msg.get("sources"):
            with st.expander(f"View Retrieved Sources ({len(msg['sources'])} chunks)"):
                for src in msg["sources"]:
                    st.markdown(f"**{src['article']}** — {src['section']}")
                    st.text(src['text'][:300])

# Chat input
if prompt := st.chat_input(f"Ask a question about {doc_name}..."):
    # Add user message
    st.session_state.messages.append({"role": "user", "content": prompt})
    with st.chat_message("user"):
        st.markdown(prompt)
    
    # Get assistant response via streaming
    with st.chat_message("assistant"):
        placeholder = st.empty()
        full_text = ""
        
        # Progressively display tokens as they arrive
        try:
            response = requests.post(
                f"{API_BASE}/api/v1/chat/",
                json={"question": prompt, "document_id": doc_id},
                stream=True,
                timeout=60,
            )
            response.raise_for_status()
            
            for chunk in response.iter_content(chunk_size=None, decode_unicode=True):
                if chunk:
                    full_text += chunk
                    placeholder.markdown(full_text + "▌")
            
            placeholder.markdown(full_text)
            
            # Fetch sources after streaming completes
            src_resp = requests.get(f"{API_BASE}/api/v1/chat/sources", timeout=5)
            sources = []
            if src_resp.status_code == 200:
                sources = src_resp.json().get("data", {}).get("sources", [])
            
            # Show sources expander
            if sources:
                with st.expander(f"View Retrieved Sources ({len(sources)} chunks)"):
                    for src in sources:
                        st.markdown(f"**{src['article']}** — {src['section']}")
                        st.text(src['text'][:300])
            
            # Store in session state
            st.session_state.messages.append({
                "role": "assistant",
                "content": full_text,
                "sources": sources,
            })
            
        except requests.exceptions.RequestException as e:
            st.error(f"Failed to get answer. Please try again.")
            st.session_state.messages.pop()  # Remove failed user message
```

import re

def extract_citations(text: str) -> list[dict]:
    """Extract [Article X, Section Y] patterns from text."""
    pattern = r'\[(Article\s+[^,]+),\s*(Section\s+[^\]]+)\]'
    matches = re.findall(pattern, text)
    return [{"article": m[0], "section": m[1]} for m in matches]

def render_citation_marker(article: str, section: str) -> str:
    """Return markdown-formatted citation badge."""
    return f'**[{article}, {section}]({article})**'

def render_sources_expander(sources: list[dict]) -> str:
    """Format sources list for display."""
    parts = []
    for s in sources:
        article = s.get("article", "N/A")
        section = s.get("section", "")
        text = s.get("text", "")[:300]
        parts.append(f"**{article}** — {section}\n\n{text}\n\n---")
    return "\n\n".join(parts)

### Step D: Create `app/components/doc_preview.py`

Tree rendering utility for document structure preview:

```python
def render_article_tree(chunks: list[dict]) -> list[dict]:
    """Convert flat chunk list to hierarchical tree structure.
    
    Groups chunks by article, then section, then subsection.
    Returns a nested dict structure for Streamlit expander rendering.
    """
    tree = {}
    for chunk in chunks:
        article = chunk.get("article", "General")
        section = chunk.get("section", "")
        subsection = chunk.get("subsection", "")
        
        if article not in tree:
            tree[article] = {"sections": {}}
        
        if section:
            if section not in tree[article]["sections"]:
                tree[article]["sections"][section] = {"chunks": []}
            tree[article]["sections"][section]["chunks"].append(chunk)
        else:
            tree[article].setdefault("chunks", []).append(chunk)
    
    return tree
```

        chunks = data.get("data", {}).get("chunks", [])

        articles = {}
        for chunk in chunks:
            article = chunk.get("article") or "General"
            if article not in articles:
                articles[article] = []
            articles[article].append(chunk)
        
        # Render hierarchical tree
        for article_name, article_chunks in articles.items():
            with st.expander(f"**{article_name}**", expanded=False):
                for chunk in article_chunks:
                    section = chunk.get("section")
                    text = chunk.get("text", "")[:500]
                    if section:

                    st.text(text)
                    st.divider()
    else:
        st.error("Could not load document preview. The document may have been removed.")
        
except requests.exceptions.RequestException:
