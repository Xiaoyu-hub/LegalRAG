    **Step 1 — Create backend/pipeline/structure_extractor.py** with:
    ```python
    import re
    from typing import Optional

    class StructureExtractor:
        PATTERNS = {
            "article": re.compile(r"^(Article)\s+((?:[IVXLCDM]+|\d+))", re.IGNORECASE),
            "section": re.compile(r"^(Section)\s+(\d+)", re.IGNORECASE),
            "subsection_letter": re.compile(r"^\(([a-z])\)"),
            "subsection_num": re.compile(r"^\((\d+)\)"),
            "definition": re.compile(r'"([^"]+)"\s+means\s+'),
        }
        
        def extract(self, text: str) -> list[dict]:
            """Process text line-by-line, return hierarchy list.
            
            Each dict in returned list:
            {"type": "article"|"section"|"subsection",
             "number": str,        # article/section number
             "label": str,         # subsection letter/number  
             "text": str,          # full line text
             "parent_article": str | None,
             "parent_section": str | None}
            """
            lines = text.split("\n")
            hierarchy = []
            current_article = None
            current_section = None
            
            for line in lines:
                stripped = line.strip()
                if not stripped:
                    continue
                    
                if m := self.PATTERNS["article"].match(stripped):
                    current_article = m.group(2).strip()
                    current_section = None
                    hierarchy.append({
                        "type": "article",
                        "number": current_article,
                        "text": stripped,
                        "parent_article": None,
                        "parent_section": None
                    })
                elif m := self.PATTERNS["section"].match(stripped):
                    current_section = m.group(2)
                    hierarchy.append({
                        "type": "section",
                        "number": current_section,
                        "text": stripped,
                        "parent_article": current_article,
                        "parent_section": None
                    })
                elif current_section and (m := self.PATTERNS["subsection_letter"].match(stripped)):
                    hierarchy.append({
                        "type": "subsection",
                        "label": m.group(1),
                        "text": stripped,
                        "parent_article": current_article,
                        "parent_section": current_section
                    })
                elif current_section and (m := self.PATTERNS["subsection_num"].match(stripped)):
                    hierarchy.append({
                        "type": "subsection",
                        "label": m.group(1),
                        "text": stripped,
                        "parent_article": current_article,
                        "parent_section": current_section
                    })
                else:
                    # Body text — attach to current section/article
                    if current_section:
                        hierarchy.append({
                            "type": "body_text",
                            "text": stripped,
                            "parent_article": current_article,
                            "parent_section": current_section
                        })
                    elif current_article:
                        hierarchy.append({
                            "type": "body_text",
                            "text": stripped,
                            "parent_article": current_article,
                            "parent_section": None
                        })
            
            return hierarchy

        def extract_definitions(self, text: str) -> list[dict]:
            """Find legal term definitions like 'Person' means..."""
            definitions = []
            for m in self.PATTERNS["definition"].finditer(text):
                term = m.group(1)
                # Get the rest of the sentence after "means"
                start = m.end()
                end = text.find(".", start)
                end = len(text) if end == -1 else end + 1
                definition_text = text[start:end].strip()
                definitions.append({
                    "term": term,
                    "definition": definition_text,
                    "context": text[max(0, m.start()-80):end].strip()
                })
            return definitions
    ```

    from typing import Optional
    from backend.models.chunk import Chunk

    class LegalChunker:
        def __init__(self, strategy: str = "by_section"):
            # Per D-07: Default to "by_section"
            # Per D-08: Support "by_article" as configurable alternative
            if strategy not in ("by_section", "by_article"):
                raise ValueError(f"Unknown strategy: {strategy}")
            self.strategy = strategy

        def chunk(self, structure: list[dict], text: str = "") -> list[Chunk]:
            if self.strategy == "by_section":
                return self._chunk_by_section(structure)
            return self._chunk_by_article(structure)

        def _chunk_by_section(self, structure: list[dict]) -> list[Chunk]:
            """Each Section becomes a chunk. Subsections and body text within that section 
            are collected into the chunk text. Per D-10, target 512-1024 tokens.
            
            Per D-09: Preserve full hierarchy: article, section, subsection, parent_article.
            Chunk metadata dict shape (from architecture.md section 3.2):
            {chunk_id: str, article: str, section: str, subsection: str, 
             text: str, title: str, parent_article: str}
            """
            chunks = []
            current_section_data = None
            section_texts = []
            chunk_index = 0

            for element in structure:
                if element["type"] == "article":
                    continue  # Articles are handled via parent_article on sections
                elif element["type"] == "section":
                    # Save previous section if exists
                    if current_section_data:
                        chunks.append(self._make_chunk(
                            current_section_data, section_texts, chunk_index
                        ))
                        chunk_index += 1
                    current_section_data = element
                    section_texts = [element["text"]]

                ))
            
            return chunks

    import sqlite3
    from typing import Optional
    from backend.models.document import Document

    class MetadataStore:
        def __init__(self, db_path: str = "data/metadata.db"):
            self.db_path = db_path
            self._init_db()

        def _init_db(self):
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.executescript("""
                CREATE TABLE IF NOT EXISTS documents (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    filename TEXT NOT NULL,
                    title TEXT DEFAULT '',
                    page_count INTEGER DEFAULT 0,
                    file_path TEXT DEFAULT '',
                    file_size INTEGER DEFAULT 0,
                    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
                );

                CREATE TABLE IF NOT EXISTS chunks (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    doc_id INTEGER NOT NULL,
                    chunk_index INTEGER DEFAULT 0,
                    text TEXT NOT NULL,
                    article TEXT,
                    section TEXT,
                    subsection TEXT,
                    title TEXT DEFAULT '',
                    parent_article TEXT,
                    strategy TEXT DEFAULT 'by_section',
                    FOREIGN KEY (doc_id) REFERENCES documents(id) ON DELETE CASCADE
                );

                CREATE TABLE IF NOT EXISTS definitions (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    doc_id INTEGER NOT NULL,
                    term TEXT NOT NULL,
                    definition TEXT NOT NULL,
                    context TEXT,
                    FOREIGN KEY (doc_id) REFERENCES documents(id) ON DELETE CASCADE
                );

                CREATE INDEX IF NOT EXISTS idx_chunks_doc_id ON chunks(doc_id);
                CREATE INDEX IF NOT EXISTS idx_definitions_doc_id ON definitions(doc_id);
            """)
            conn.commit()
            conn.close()

        def insert_document(self, doc: Document) -> int:
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute(
                "INSERT INTO documents (filename, title, page_count, file_path, file_size) VALUES (?, ?, ?, ?, ?)",
                (doc.filename, doc.title, doc.page_count, doc.file_path, doc.file_size)
            )
            conn.commit()
            doc_id = c.lastrowid
            conn.close()
            return doc_id

            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            for chunk in chunks:
                c.execute(
                    """INSERT INTO chunks 
                       (doc_id, chunk_index, text, article, section, subsection, title, parent_article, strategy)
                       VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)""",
                    (doc_id, chunk.chunk_index, chunk.text, chunk.article,
                     chunk.section, chunk.subsection, chunk.title,
                     chunk.parent_article, chunk.strategy)
                )
            conn.commit()
            conn.close()

        def insert_definitions(self, doc_id: int, definitions: list[dict]):
            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            for d in definitions:
                c.execute(
                    "INSERT INTO definitions (doc_id, term, definition, context) VALUES (?, ?, ?, ?)",

            conn.close()

            if row:

            rows = c.fetchall()
            conn.close()

        def get_chunk_texts(self) -> list[str]:
            return [c.text for c in self.get_all_chunks()]

            conn = sqlite3.connect(self.db_path)
            c = conn.cursor()
            c.execute("SELECT COUNT(*) FROM chunks")
