    **Step 1 — Create backend/retrieval/hybrid_search.py**:
    ```python
    """Hybrid search combining BM25 and vector search with score normalization.
    
    Per D-15: Alpha weighting 0.7 vector / 0.3 BM25.
    Per D-18: Return top-10 before reranking.
    
    CRITICAL (Pitfall 1): BM25 and FAISS scores are on incompatible scales.
    Min-max normalize BOTH per query before weighted combination.
    """
    import numpy as np
    from backend.retrieval.bm25 import BM25Retriever
    from backend.retrieval.vector_search import VectorSearch
    
    def normalize_scores(scores: np.ndarray) -> np.ndarray:
        """Min-max normalize scores to [0, 1].
        Handles edge case where all scores are equal (returns zeros).
        """
        min_s, max_s = scores.min(), scores.max()
        if max_s == min_s:
            return np.zeros_like(scores, dtype=np.float32)
        return (scores - min_s) / (max_s - min_s)
    
    class HybridSearch:
        def __init__(self, alpha: float = 0.7, top_k: int = 10):
            # Per D-15: default 0.7 vector, 0.3 BM25
            self.alpha = alpha
            self.top_k = top_k
            self.vector_search = VectorSearch.get_instance()
            self.bm25 = BM25Retriever.get_instance()
        
        def search(self, query_vec: np.ndarray, tokenized_query: list[str], 
                  alpha: float | None = None, top_k: int | None = None) -> list[tuple[int, float]]:
            """Hybrid search: vector + BM25 with min-max normalized score fusion.
            
            Args:
                query_vec: L2-normalized query embedding (1024,)
                tokenized_query: Tokenized query for BM25
                alpha: Vector weight (default: self.alpha)
                top_k: Number of results to return (default: self.top_k)
            Returns:
                List of (chunk_index, hybrid_score) tuples sorted by score descending
            """
            alpha = alpha if alpha is not None else self.alpha
            k = top_k if top_k is not None else self.top_k
            
            n_total = self.vector_search.size
            if n_total == 0:
                return []
            
            # 1. Vector search — get top-(k*2) for better recall before normalization
            vec_distances, vec_indices = self.vector_search.search(query_vec, k=k*2)
            
            # 2. BM25 scores for all documents
            bm25_scores = self.bm25.get_scores(tokenized_query)
            
            # 3. Build per-document vector score array
            vec_scores = np.zeros(n_total, dtype=np.float32)
            for dist, idx in zip(vec_distances[0], vec_indices[0]):
                if idx != -1:
                    vec_scores[idx] = dist
            
            # 4. Min-max normalize BOTH score distributions (Pitfall 1)
            vec_norm = normalize_scores(vec_scores)
            bm25_norm = normalize_scores(bm25_scores)
            
            # 5. Weighted fusion
            # Per architecture.md section 3.3: score = alpha * vector + (1-alpha) * bm25
            hybrid = alpha * vec_norm + (1.0 - alpha) * bm25_norm
            
            # 6. Get top-K indices
            top_indices = np.argsort(hybrid)[::-1][:k]
            return [(int(idx), float(hybrid[idx])) for idx in top_indices]
    ```

    **Step 2 — Create backend/retrieval/reranker.py**:
    ```python
    """Cross-encoder reranker for improving top-K result quality.
    
    Per D-17: cross-encoder/ms-marco-MiniLM-L-6-v2 (lightweight, CPU-friendly).
    Per D-18: Takes top-10 from hybrid, returns top-5 after reranking.
    
    CRITICAL (Pitfall 4): Cross-encoder has 512-token max length.
    Chunks exceeding this get truncated by the tokenizer.
    """
    from sentence_transformers import CrossEncoder
    
    class Reranker:
        _instance = None
        
        def __init__(self, model_name: str = "cross-encoder/ms-marco-MiniLM-L-6-v2"):
            if Reranker._instance is not None:
                raise RuntimeError("Use Reranker.get_instance() instead")
            self.model = CrossEncoder(model_name, max_length=512)
            self.model_name = model_name
        
        @classmethod
        def get_instance(cls, model_name: str = "cross-encoder/ms-marco-MiniLM-L-6-v2"):
            if cls._instance is None:
                cls._instance = cls(model_name)
            return cls._instance
        
        def rerank(self, query: str, passages: list[str], top_k: int = 5) -> list[dict]:
            """Rank passages by relevance to query using cross-encoder.
            
            Args:
                query: The user's search query
                passages: List of passage text strings to rerank
                top_k: Number of top results to return
            Returns:
                List of dicts: {"corpus_id": int, "score": float, "text": str}
                Sorted by score descending.
            """
            if not passages:
                return []
            
            results = self.model.rank(query, passages, top_k=top_k, return_documents=True)
            return [
                {
                    "corpus_id": r["corpus_id"],
                    "score": float(r["score"]),
                    "text": r["text"]
                }
                for r in results
            ]
    ```
