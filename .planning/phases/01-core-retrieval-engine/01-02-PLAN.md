    """Embedder using BGE-large-en-v1.5 via SentenceTransformer.
    
    Per D-11: BGE-large-en-v1.5 embedding model.
    Per D-12: batch_size=32 for embedding generation.
    Singleton pattern - one model instance shared across the app.
    """
    import numpy as np
    from sentence_transformers import SentenceTransformer
    
    # Per BGE specification: for short query to long passage (s2p)
    QUERY_INSTRUCTION = "Represent this sentence for searching relevant passages: "
    
    class Embedder:
        _instance = None
        
        def __init__(self, model_name: str = "BAAI/bge-large-en-v1.5"):
            if Embedder._instance is not None:
                raise RuntimeError("Use Embedder.get_instance() instead")
            self.model = SentenceTransformer(model_name)
            self.model_name = model_name
        
        @classmethod
        def get_instance(cls, model_name: str = "BAAI/bge-large-en-v1.5"):
            if cls._instance is None:
                cls._instance = cls(model_name)
            return cls._instance
        
        def encode(self, texts: list[str]) -> np.ndarray:
            """Encode passages (no instruction prepended).
            Returns float32 array of shape (n_texts, 1024) with L2-normalized vectors.
            """
            return self.model.encode(
                texts,
                normalize_embeddings=True,
                batch_size=32  # Per D-12
            )
        
        def encode_query(self, query: str) -> np.ndarray:
            """Encode query with instruction prepended.
            Returns float32 array of shape (1024,) with L2-normalized vector.
            """
            return self.model.encode(
                [QUERY_INSTRUCTION + query],
                normalize_embeddings=True,
                batch_size=1
            )[0]
        
        @property
        def dimension(self) -> int:
            return self.model.get_sentence_embedding_dimension()
    ```

    """BM25 keyword retrieval using rank_bm25.
    
    Per D-16: Use rank_bm25 library for keyword retrieval.
    """
    import re
    import numpy as np
    from rank_bm25 import BM25Okapi
    
    class BM25Retriever:
        _instance = None
        
        def __init__(self):
            if BM25Retriever._instance is not None:
                raise RuntimeError("Use BM25Retriever.get_instance() instead")
            self.bm25 = None
            self.chunk_ids = []
            self.corpus = []
        
        @classmethod
        def get_instance(cls):
            if cls._instance is None:
                cls._instance = cls()
            return cls._instance
        
        @staticmethod
        def tokenize(text: str) -> list[str]:
            """Tokenize text for BM25. Lowercase, split on word boundaries.
            Per RESEARCH.md Pitfall 5: Define a single tokenize() used for both
            corpus building and querying. Use same tokenizer consistently.
            """
            text = text.lower()
            return re.findall(r'\w+', text)
        
        def build_index(self, texts: list[str], chunk_ids: list[int] | None = None):
            """Build BM25 index from chunk texts.
            Args:
                texts: List of chunk text strings
                chunk_ids: Corresponding chunk IDs (default: range(len(texts)))
            """
            self.corpus = texts
            self.chunk_ids = chunk_ids if chunk_ids is not None else list(range(len(texts)))
            tokenized_corpus = [self.tokenize(t) for t in texts]
            self.bm25 = BM25Okapi(tokenized_corpus)
        
        def search(self, query: str, n: int = 10) -> list[tuple[int, float]]:
            """Search BM25 index, return top-n (chunk_id, score) pairs."""
            if self.bm25 is None:
                raise RuntimeError("BM25 index not built. Call build_index() first.")
            tokenized_query = self.tokenize(query)
            if not tokenized_query:
                return []
            scores = self.bm25.get_scores(tokenized_query)
            top_indices = np.argsort(scores)[::-1][:n]
            return [(self.chunk_ids[i], float(scores[i])) for i in top_indices]
        
        def get_scores(self, tokenized_query: list[str]) -> np.ndarray:
            """Get BM25 scores for all documents. Used by hybrid search."""
            if self.bm25 is None:
                raise RuntimeError("BM25 index not built.")
            return np.array(self.bm25.get_scores(tokenized_query))
        
        def reset(self):
            """Clear the index. Used when re-indexing."""
            self.bm25 = None
            self.chunk_ids = []
            self.corpus = []
    ```

    ```python
    """FAISS vector similarity search.
    
    Per D-13: FAISS with IndexFlatIP for exact similarity search.
    Uses L2-normalized vectors (cosine similarity via inner product).
    """
    import os
    import warnings
    import numpy as np
    
    class VectorSearch:
        _instance = None
        
        def __init__(self, dimension: int = 1024):
            if VectorSearch._instance is not None:
                raise RuntimeError("Use VectorSearch.get_instance() instead")
            self.dimension = dimension
            self._init_index()
            self.chunk_ids = []  # Maps FAISS index position -> chunk_id
        
        def _init_index(self):
            import faiss
            self.index = faiss.IndexFlatIP(self.dimension)
        
        @classmethod
        def get_instance(cls, dimension: int = None):
            if cls._instance is None:
                if dimension is None:
                    dimension = 1024  # BGE-large-en-v1.5 default
                cls._instance = cls(dimension)
            return cls._instance
        
        def add(self, embeddings: np.ndarray, chunk_ids: list[int] | None = None):
            """Add embeddings to the index.
            
            Automatically L2-normalizes if vectors are not unit length.
            Per D-13 and Pitfall 2: All vectors MUST be unit-length for IndexFlatIP.
            """
            import faiss
            # Check if normalization needed
            norms = np.linalg.norm(embeddings, axis=1)
            if not np.allclose(norms, 1.0, atol=1e-5):
                warnings.warn("Embeddings not normalized. Auto-normalizing.")
                embeddings = embeddings.copy()
                faiss.normalize_L2(embeddings)
            
            self.index.add(embeddings)
            if chunk_ids:
                start = len(self.chunk_ids)
                self.chunk_ids.extend(chunk_ids)
            else:
                self.chunk_ids.extend(range(start, start + embeddings.shape[0]))
        
        def search(self, query_vec: np.ndarray, k: int = 10) -> tuple[np.ndarray, np.ndarray]:
            """Search index for nearest neighbors.
            
            Args:
                query_vec: L2-normalized query vector of shape (dim,)
                k: Number of results to return
            Returns:
                (distances, indices): FAISS result arrays
            """
            import faiss
            # Ensure query is 2D and normalized
            query_2d = query_vec.reshape(1, -1).astype(np.float32)
            norms = np.linalg.norm(query_2d, axis=1)
            if not np.allclose(norms, 1.0, atol=1e-5):
                warnings.warn("Query vector not normalized. Auto-normalizing.")
                faiss.normalize_L2(query_2d)
            
            distances, indices = self.index.search(query_2d, k)
            return distances, indices
        
        def get_chunk_ids_for_indices(self, indices: np.ndarray) -> list[int]:
            """Map FAISS index positions to chunk IDs."""
            return [self.chunk_ids[i] if i != -1 else -1 for i in indices[0]]
        
        def save(self, path: str):
            """Save FAISS index to file."""
            import faiss
            os.makedirs(os.path.dirname(path), exist_ok=True)
            # Write to temp file then rename for atomicity (Pitfall 6)
            tmp_path = path + ".tmp"
            faiss.write_index(self.index, tmp_path)
            os.replace(tmp_path, path)
        
        def load(self, path: str, chunk_ids: list[int] | None = None):
            """Load FAISS index from file."""
            import faiss
            if not os.path.exists(path):
                raise FileNotFoundError(f"Index file not found: {path}")
            self.index = faiss.read_index(path)
            self.dimension = self.index.d
            if chunk_ids:
                self.chunk_ids = chunk_ids
            else:
                self.chunk_ids = list(range(self.index.ntotal))
        
        @property
        def size(self) -> int:
            return self.index.ntotal
        
        def reset(self):
            """Clear the index."""
            self._init_index()
            self.chunk_ids = []
    ```
